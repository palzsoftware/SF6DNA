       AND m.updated_at IS NOT DISTINCT FROM a.updated_at) <> 661
 THEN RAISE EXCEPTION 'Rollback drift/missing backup: abort; do not overwrite later edits'; END IF;
END $$;
UPDATE public.moves m SET status='draft' FROM retained_after a
 WHERE m.id=a.id AND m.status='published' AND m.updated_at IS NOT DISTINCT FROM a.updated_at
 RETURNING m.id,m.status,m.updated_at;

 IF (SELECT count(*) FROM public.moves m JOIN approved_rollback_ids t USING(id) WHERE m.status='draft') <> 661
 THEN RAISE EXCEPTION 'Rollback count mismatch'; END IF;
END $$;
